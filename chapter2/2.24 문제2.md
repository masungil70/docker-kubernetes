기존 backend 코드

main.py

```python

from fastapi import FastAPI, HTTPException
import os
import socket
import signal
import sys
from datetime import datetime

app = FastAPI()

# 환경변수
APP_MODE = os.getenv("APP_MODE", "development")
APP_VERSION = os.getenv("APP_VERSION", "unknown")

# hostname (컨테이너 구분용)
HOSTNAME = socket.gethostname()

# 샘플 데이터
users = [
    {"id": 1, "name": "kim", "role": "admin", "created_at": "2026-01-01T10:00:00Z"},
    {"id": 2, "name": "hong", "role": "user", "created_at": "2026-01-02T10:00:00Z"},
    {"id": 3, "name": "lee", "role": "user", "created_at": "2026-01-03T10:00:00Z"},
]


# ---------------------------
# 종료 시그널 처리
# ---------------------------
def handle_sigterm(signum, frame):
    print("Received SIGTERM")
    print("Shutting down gracefully...")
    # 필요한 cleanup 작업 수행
    print("Cleanup complete")
    sys.exit(0)


# ---------------------------
# Health Check
# ---------------------------
@app.get("/health")
def health():
    return {
        "status": "ok",
        "service": "api-server",
        "instance": HOSTNAME,
        "timestamp": datetime.utcnow().isoformat()
    }


# ---------------------------
# 전체 사용자 조회
# ---------------------------
@app.get("/api/users")
def get_users():
    return {
        "count": len(users),
        "users": users
    }


# ---------------------------
# 사용자 단건 조회
# ---------------------------
@app.get("/api/users/{user_id}")
def get_user(user_id: int):
    for user in users:
        if user["id"] == user_id:
            return user

    raise HTTPException(
        status_code=404,
        detail={
            "error": "User not found",
            "id": user_id
        }
    )


# ---------------------------
# 환경 설정 확인
# ---------------------------
@app.get("/api/config")
def get_config():
    return {
        "mode": APP_MODE,
        "version": APP_VERSION,
        "hostname": HOSTNAME
    }
```
